function login(){

const username=
document.getElementById('username')
.value.trim();

const password=
document.getElementById('password').value;

const msg=
document.getElementById('loginMsg');

if(!username||!password){

msg.textContent='Username dan password wajib diisi.';

return;

}

msg.textContent='Memeriksa login...';

google.script.run

.withSuccessHandler(r=>{

if(r.success){

document.getElementById('loginPage')
.style.display='none';

document.body.classList.remove('login-mode');
document.body.classList.add('app-mode');

document.getElementById('app')

.textContent=r.user.nama;

}else{

msg.textContent=r.message;

}

})

.withFailureHandler(e=>{

msg.textContent='Koneksi gagal.';

console.error(e);

})

.checkLogin(username,password);

}

function displayNumberRaw(value){

  if(value === undefined || value === null || value === ''){
    return '-';
  }

  /*
   * Jangan gunakan toFixed(), parseFloat(), atau Number() di tampilan.
   * Tujuannya mempertahankan tampilan angka sebagaimana data disimpan.
   */
  return String(value);

}

function generateRoomId(){
  const d=new Date();
  const pad=n=>String(n).padStart(2,'0');
  return 'RNG-'+d.getFullYear()+pad(d.getMonth()+1)+pad(d.getDate())+'-'+
    pad(d.getHours())+pad(d.getMinutes())+pad(d.getSeconds())+'-'+
    Math.random().toString(36).substring(2,7).toUpperCase();
}

function renderInlineQr(targetId,text,size=170){
  const target=document.getElementById(targetId);
  if(!target)return;
  target.innerHTML='';
  if(!text){
    target.innerHTML='<div class="empty">ID belum tersedia.</div>';
    return;
  }
  if(typeof QRCode!=='undefined'){
    try{
      new QRCode(target,{
        text:String(text),
        width:size,
        height:size,
        correctLevel:QRCode.CorrectLevel.M
      });
      return;
    }catch(e){}
  }
  target.innerHTML=`<img style="width:${size}px;height:${size}px;object-fit:contain" alt="QR Code" src="https://quickchart.io/qr?text=${encodeURIComponent(text)}&size=500&margin=2&ecLevel=M">`;
}

function trxDateValue(v){
  if(!v)return '';
  const str=String(v).trim();
  const m=str.match(/(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})/);
  if(m)return `${m[1]}-${String(m[2]).padStart(2,'0')}-${String(m[3]).padStart(2,'0')}`;
  const d=str.match(/(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})/);
  if(d)return `${d[3]}-${String(d[2]).padStart(2,'0')}-${String(d[1]).padStart(2,'0')}`;
  return '';
}

function filterLaporanTransaksi(){
  const body=document.getElementById('trxReportBody');
  if(!body)return;
  const dari=document.getElementById('trxFilterDari')?.value||'';
  const sampai=document.getElementById('trxFilterSampai')?.value||'';
  const jenis=(document.getElementById('trxFilterJenis')?.value||'').toLowerCase();
  const cari=(document.getElementById('trxFilterCari')?.value||'').trim().toLowerCase();

  const semua=Array.isArray(appData.transaksi)?appData.transaksi:[];
  const data=semua.filter(t=>{
    const d=trxDateValue(t.Tanggal);
    if(dari && (!d || d<dari))return false;
    if(sampai && (!d || d>sampai))return false;
    if(jenis && String(t.Jenis||'').toLowerCase()!==jenis)return false;
    if(cari){
      const hay=[t.ID_Transaksi,t.ID_Barang,t.Nama_Barang,t.Keterangan].map(x=>String(x||'').toLowerCase()).join(' ');
      if(!hay.includes(cari))return false;
    }
    return true;
  });

  const totalMasuk=data.filter(t=>String(t.Jenis||'').toLowerCase()==='masuk').reduce((n,t)=>n+(Number(t.Jumlah)||0),0);
  const totalKeluar=data.filter(t=>String(t.Jenis||'').toLowerCase()==='keluar').reduce((n,t)=>n+(Number(t.Jumlah)||0),0);

  body.innerHTML=`
  <div class="table-box">
    <div style="padding:12px 8px 8px;text-align:center">
      <h1 style="margin:0 0 4px;font-size:18px">LAPORAN TRANSAKSI INVENTARIS</h1>
      <p style="margin:0;font-size:11px">Riwayat barang masuk dan barang keluar</p>
    </div>
    <div style="display:flex;gap:10px;flex-wrap:wrap;margin:0 0 10px;padding:0 8px">
      <div style="padding:8px 12px;border:1px solid #bbf7d0;border-radius:8px;background:#f0fdf4"><b>Total Masuk:</b> ${totalMasuk}</div>
      <div style="padding:8px 12px;border:1px solid #fecaca;border-radius:8px;background:#fef2f2"><b>Total Keluar:</b> ${totalKeluar}</div>
      <div style="padding:8px 12px;border:1px solid #cbd5e1;border-radius:8px;background:#f8fafc"><b>Jumlah Transaksi:</b> ${data.length}</div>
    </div>
    <table>
      <thead><tr>
        <th>No</th><th>ID Transaksi</th><th>Tanggal</th><th>Jenis</th>
        <th>ID Barang</th><th>Nama Barang</th><th>Jumlah</th><th>Keterangan</th>
      </tr></thead>
      <tbody>
        ${data.length?data.map((t,i)=>`<tr>
          <td>${i+1}</td>
          <td>${esc(t.ID_Transaksi||'-')}</td>
          <td>${esc(t.Tanggal||'-')}</td>
          <td>${esc(t.Jenis||'-')}</td>
          <td>${esc(t.ID_Barang||'-')}</td>
          <td style="text-align:left">${esc(t.Nama_Barang||'-')}</td>
          <td>${t.Jumlah??0}</td>
          <td style="text-align:left">${esc(t.Keterangan||'-')}</td>
        </tr>`).join(''):'<tr><td colspan="8">Tidak ada transaksi sesuai filter.</td></tr>'}
      </tbody>
    </table>
  </div>`;
}

function resetFilterLaporanTransaksi(){

  const j=document.getElementById('trxFilterJenis');if(j)j.value='';
  filterLaporanTransaksi();
}

let qrStream = null;
let qrScannerRunning = false;
let qrScanTimer = null;
let lastQrValue = '';
let lastQrTime = 0;

function setQrStatus(message, ok=false){
  const el=document.getElementById('qrScanStatus');
  if(!el) return;
  el.innerHTML=(ok?'✅ ':'')+esc(message);
}

async function startQrScanner(){
  const video=document.getElementById('qrVideo');
  const canvas=document.getElementById('qrCanvas');
  if(!video || !canvas) return;
  if(qrScannerRunning) return;

  if(!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia){
    setQrStatus('Browser tidak mendukung kamera. Gunakan Chrome/Edge terbaru atau tombol Scan dari Foto.');
    return;
  }

  if(typeof jsQR==='undefined'){
    setQrStatus('Library pembaca QR belum termuat. Periksa internet lalu refresh halaman.');
    return;
  }

  try{
    stopQrScanner();
    setQrStatus('Meminta izin kamera...');

    // Minta kamera belakang terlebih dahulu. Jika tidak tersedia, gunakan kamera apa pun.
    let stream;
    try{
      stream=await navigator.mediaDevices.getUserMedia({
        video:{facingMode:{ideal:'environment'},width:{ideal:1280},height:{ideal:720}},
        audio:false
      });
    }catch(firstError){
      stream=await navigator.mediaDevices.getUserMedia({video:true,audio:false});
    }

    qrStream=stream;
    video.srcObject=stream;
    video.setAttribute('playsinline','true');
    video.muted=true;
    await video.play();
    qrScannerRunning=true;
    setQrStatus('Kamera aktif. Arahkan QR ke kotak di tengah.', true);

    const ctx=canvas.getContext('2d',{willReadFrequently:true});
    const scan=()=>{
      if(!qrScannerRunning || video.readyState < 2){
        qrScanTimer=requestAnimationFrame(scan);
        return;
      }
      const w=video.videoWidth;
      const h=video.videoHeight;
      if(w && h){
        canvas.width=w;
        canvas.height=h;
        ctx.drawImage(video,0,0,w,h);
        const image=ctx.getImageData(0,0,w,h);
        const code=jsQR(image.data,w,h,{inversionAttempts:'attemptBoth'});
        if(code && code.data){
          const now=Date.now();
          if(code.data!==lastQrValue || now-lastQrTime>2500){
            lastQrValue=code.data;
            lastQrTime=now;
            processQrScan(code.data);
            setQrStatus('QR terbaca: '+code.data, true);
          }
        }
      }
      qrScanTimer=requestAnimationFrame(scan);
    };
    qrScanTimer=requestAnimationFrame(scan);
  }catch(err){
    console.error('CAMERA ERROR',err);
    qrScannerRunning=false;
    if(qrStream){qrStream.getTracks().forEach(t=>t.stop());qrStream=null;}
    let msg=err && err.message ? err.message : String(err||'Kesalahan kamera');
    if(err && (err.name==='NotAllowedError' || /permission|denied/i.test(msg))){
      msg='Izin kamera ditolak. Klik ikon gembok di alamat browser → Kamera → Izinkan, lalu refresh halaman.';
    }else if(err && (err.name==='NotFoundError' || /not found|no camera/i.test(msg))){
      msg='Kamera tidak ditemukan. Pastikan kamera terpasang dan tidak sedang dipakai aplikasi lain.';
    }else if(err && err.name==='NotReadableError'){
      msg='Kamera sedang digunakan aplikasi lain. Tutup aplikasi kamera/Zoom/Meet lalu coba lagi.';
    }else if(location.protocol!=='https:' && location.hostname!=='localhost'){
      msg='Kamera membutuhkan HTTPS. Buka Web App menggunakan URL /exec Google Apps Script.';
    }
    setQrStatus(msg);
  }
}

function stopQrScanner(){
  qrScannerRunning=false;
  if(qrScanTimer){cancelAnimationFrame(qrScanTimer);qrScanTimer=null;}
  if(qrStream){
    qrStream.getTracks().forEach(track=>{try{track.stop();}catch(e){}});
    qrStream=null;
  }
  const video=document.getElementById('qrVideo');
  if(video){
    try{video.pause();}catch(e){}
    video.srcObject=null;
  }
  const status=document.getElementById('qrScanStatus');
  if(status) status.textContent='Kamera dihentikan.';
}

function processQrScan(value){
  const status=document.getElementById('qrScanStatus');
  const raw=String(value||'').trim();
  if(!raw)return;

  // 1. Coba QR Prasarana terlebih dahulu.
  if(status) status.textContent='QR terbaca. Mencari prasarana dan sarana...';
  google.script.run
    .withSuccessHandler(roomResult=>{
      if(roomResult && roomResult.success && roomResult.ruangan){
        if(status) status.textContent='Prasarana ditemukan: '+roomResult.ruangan.Nama_Ruangan;
        showPrasaranaScanResult(roomResult);
        stopQrScanner();
        return;
      }

      // 2. Kalau bukan QR prasarana, coba sebagai QR barang.
      const barang=getBarangFromQrValue(raw);
      if(barang){
        if(status) status.textContent='Barang ditemukan: '+barang.Nama_Barang;
        showQrResult(barang);
        stopQrScanner();
        return;
      }

      google.script.run
        .withSuccessHandler(r=>{
          if(r && r.success && r.barang){
            showQrResult(r.barang);
            if(status) status.textContent='Barang ditemukan: '+r.barang.Nama_Barang;
            stopQrScanner();
          }else{
            if(status) status.textContent='QR terbaca, tetapi data tidak ditemukan.';
            showQrResult(null,raw);
          }
        })
        .withFailureHandler(e=>{
          if(status) status.textContent='Gagal mencari data QR di server.';
          showQrResult(null,raw);
          console.error(e);
        })
        .getBarangByQR(raw);
    })
    .withFailureHandler(e=>{
      console.error(e);
      // Jika lookup prasarana gagal, tetap coba QR barang.
      const barang=getBarangFromQrValue(raw);
      if(barang){
        showQrResult(barang);
        if(status) status.textContent='Barang ditemukan: '+barang.Nama_Barang;
        stopQrScanner();
        return;
      }
      google.script.run
        .withSuccessHandler(r=>{
          if(r && r.success && r.barang){
            showQrResult(r.barang);
            if(status) status.textContent='Barang ditemukan: '+r.barang.Nama_Barang;
            stopQrScanner();
          }else{
            if(status) status.textContent='Data QR tidak ditemukan.';
            showQrResult(null,raw);
          }
        })
        .getBarangByQR(raw);
    })
    .getPrasaranaScanData(raw);
}

function scanQrImage(input){
  const file=input && input.files ? input.files[0] : null;
  if(!file) return;
  if(typeof jsQR==='undefined'){
    setQrStatus('Library pembaca QR belum termuat.');
    return;
  }
  const img=new Image();
  const url=URL.createObjectURL(file);
  img.onload=function(){
    try{
      const canvas=document.createElement('canvas');
      const max=1800;
      const scale=Math.min(1,max/Math.max(img.width,img.height));
      canvas.width=Math.max(1,Math.floor(img.width*scale));
      canvas.height=Math.max(1,Math.floor(img.height*scale));
      const ctx=canvas.getContext('2d',{willReadFrequently:true});
      ctx.drawImage(img,0,0,canvas.width,canvas.height);
      const data=ctx.getImageData(0,0,canvas.width,canvas.height);
      const code=jsQR(data.data,canvas.width,canvas.height,{inversionAttempts:'attemptBoth'});
      if(code && code.data){
        setQrStatus('QR dari foto terbaca: '+code.data,true);
        processQrScan(code.data);
      }else{
        setQrStatus('QR tidak ditemukan pada foto. Coba foto lebih dekat dan jelas.');
      }
    }finally{
      URL.revokeObjectURL(url);
      input.value='';
    }
  };
  img.onerror=function(){
    URL.revokeObjectURL(url);
    input.value='';
    setQrStatus('Foto tidak dapat dibaca.');
  };
  img.src=url;
}

function makeQR(text,target){
  target.innerHTML='';
  if(typeof QRCode==='undefined'){target.innerHTML='<div class="empty">Library QR belum dimuat.</div>';return false;}
  try{
    // QR memuat teks data lengkap agar kamera HP menampilkan informasinya,
    // bukan hanya ID. Gunakan level koreksi M agar mudah dipindai.
    new QRCode(target,{text:String(text),width:320,height:320,correctLevel:QRCode.CorrectLevel.M});
    return true;
  }catch(e){target.innerHTML='<div class="empty">QR gagal dibuat. Kurangi panjang data atau coba lagi.</div>';return false;}
}

function showBarangQR(id){
  const b=appData.barang.find(x=>String(x.ID_Barang)===String(id));
  if(!b)return alert('Data barang tidak ditemukan.');
  const payload=buildBarangQrText(b);
  currentQrPrintData={value:String(b.ID_Barang||''),payload:payload,jenis:'QR Barang',nama:String(b.Nama_Barang||'Barang')};
  document.getElementById('qrModalTitle').textContent='🔳 QR Code Barang';
  document.getElementById('qrModalHint').textContent='Scan QR ini untuk mencari data barang.';
  document.getElementById('qrBarangName').textContent=b.Nama_Barang||'-';
  document.getElementById('qrBarangCode').textContent='ID: '+(b.ID_Barang||'-')+' | Kode: '+(b.Kode_Barang||'-');
  const t=document.getElementById('qrCodeCanvas');
  document.getElementById('qrModal').classList.add('show');
  makeQR(payload,t);
}

}

function closeQrModal(){document.getElementById('qrModal').classList.remove('show');}

function qrDataUrl(){
  const box=document.getElementById('qrCodeCanvas');
  const c=box.querySelector('canvas');
  if(c)return c.toDataURL('image/png');
  const img=box.querySelector('img');
  if(img&&img.src.startsWith('data:image/'))return img.src;
  return '';
}

function printBarangQR(){
  const q=currentQrPrintData||{};
  if(!q.value)return alert('QR belum siap.');
  const dataUrl=qrDataUrl();
  if(!dataUrl)return alert('QR belum selesai dibuat.');
  const btn=document.querySelector('#qrModal button.btn.blue'),old=btn?btn.innerHTML:'';
  if(btn){btn.disabled=true;btn.innerHTML='⏳ Menyimpan...';}
  google.script.run.withSuccessHandler(r=>{
    if(btn){btn.disabled=false;btn.innerHTML=old;}
    if(!r||!r.success)return alert(r?.message||'QR gagal disimpan.');
    document.body.classList.add('qr-printing');
    setTimeout(()=>{window.print();setTimeout(()=>document.body.classList.remove('qr-printing'),500)},150);
  }).withFailureHandler(e=>{
    if(btn){btn.disabled=false;btn.innerHTML=old;}
    alert('Gagal menyimpan QR: '+(e?.message||e));
  }).saveQrCodeToDrive(dataUrl,q.value,q.jenis,q.nama);
}
